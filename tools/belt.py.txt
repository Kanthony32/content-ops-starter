def nick(t):
    return {"Philadelphia 76ers": "76ers", "Portland Trail Blazers": "Trail Blazers"}.get(t, t.split()[-1])


def city(t):
    return CITY.get(t, " ".join(t.split()[:-1]))


def d(s):
    return dt.date.fromisoformat(s)


def fmt(s, year=True):
    x = d(s) if isinstance(s, str) else s
    return f"{x:%b} {x.day}, {x.year}" if year else f"{x:%b} {x.day}"


def load(root):
    p = os.path.join(root, "tools", "belt", "belt.json")
    return p, json.load(open(p, encoding="utf-8"))


def save(p, data):
    open(p, "w", encoding="utf-8").write(json.dumps(data, indent=2, ensure_ascii=False) + "\n")


def reigns(data):
    """Walk the games and return the list of reigns (oldest first)."""
    cur = {"n": 0, "team": data["start_holder"], "won": data["start_date"], "beat": None, "score": None,
           "defenses": [], "lost": None, "lost_to": None, "lost_score": None, "start": True}
    out = [cur]
    for g in data["games"]:
        h, a, hs, as_ = g["home"], g["away"], g["home_score"], g["away_score"]
        if cur["team"] not in (h, a):
            raise SystemExit(f"game on {g['date']} ({a} at {h}) does not involve the holder, {cur['team']}")
        opp = a if cur["team"] == h else h
        mine, theirs = (hs, as_) if cur["team"] == h else (as_, hs)
        home = cur["team"] == h
        if mine > theirs:
            cur["defenses"].append({"date": g["date"], "opp": opp, "score": f"{mine}-{theirs}", "home": home})
        else:
            cur["lost"], cur["lost_to"], cur["lost_score"] = g["date"], opp, f"{mine}-{theirs}"
            cur = {"n": len(out), "team": opp, "won": g["date"], "beat": cur["team"], "score": f"{theirs}-{mine}",
                   "defenses": [], "lost": None, "lost_to": None, "lost_score": None, "start": False,
                   "won_home": not home}
            out.append(cur)
    return out


def days(r, today):
    end = d(r["lost"]) if r["lost"] else today
    return max(0, (end - d(r["won"])).days)


# ---------------- commands ----------------
def cmd_add(a):
    p, data = load(a.root)
    for t in (a.home, a.away):
        if t not in TEAMS:
            raise SystemExit(f"unknown team {t!r}; use the full name, e.g. 'Miami Heat'")
    if a.home_score == a.away_score:
        raise SystemExit("NBA games have no draws")
    holder = reigns(data)[-1]["team"]
    if holder not in (a.home, a.away):
        raise SystemExit(f"{a.away} at {a.home} does not involve the holder, {holder}; not a belt game")
    last = data["games"][-1]["date"] if data["games"] else data["start_date"]
    if any(g["date"] == a.date and holder in (g["home"], g["away"]) for g in data["games"]):
        print(f"already recorded: {holder} game on {a.date}; nothing changed")
        return
    if d(a.date) < d(last):
        raise SystemExit(f"{a.date} is before the last recorded belt game ({last})")
    data["games"].append({"date": a.date, "home": a.home, "away": a.away,
                          "home_score": a.home_score, "away_score": a.away_score})
    save(p, data)
    now = reigns(data)[-1]["team"]
    print(("BELT RETAINED: " if now == holder else "NEW CHAMPION: ") + now)


def cmd_next(a):
    p, data = load(a.root)
    if a.none:
        data["next"] = None
    else:
        for t in (a.home, a.away):
            if t not in TEAMS:
                raise SystemExit(f"unknown team {t!r}")
        holder = reigns(data)[-1]["team"]
        if holder not in (a.home, a.away):
            raise SystemExit(f"next game must involve the holder, {holder}")
        dt.datetime.fromisoformat(a.start)
        data["next"] = {"start": a.start, "home": a.home, "away": a.away}
    save(p, data)
    print("next set:", data["next"])


def cmd_status(a):
    _, data = load(a.root)
    r = reigns(data)
    print("holder:", r[-1]["team"], "| since", r[-1]["won"], "| reigns:", len(r), "| transfers:", len(r) - 1)
    print("last game:", data["games"][-1] if data["games"] else None)
    print("next:", data.get("next"))

def main():
    ap = argparse.ArgumentParser()
    sub = ap.add_subparsers(dest="cmd", required=True)
    for name in ("add", "next", "build", "status"):
        s = sub.add_parser(name)
        s.add_argument("--root", required=True)
        if name == "add":
            s.add_argument("--date", required=True); s.add_argument("--home", required=True); s.add_argument("--away", required=True)
            s.add_argument("--home-score", type=int, required=True); s.add_argument("--away-score", type=int, required=True)
        if name == "next":
            s.add_argument("--start"); s.add_argument("--home"); s.add_argument("--away"); s.add_argument("--none", action="store_true")
        if name == "build":
            s.add_argument("--today")
    a = ap.parse_args()
    a.root = os.path.abspath(a.root)
    if a.cmd == "add":
        cmd_add(a)
    elif a.cmd == "next":
        if not a.none and not (a.start and a.home and a.away):
            sys.exit("next needs --start, --home and --away (or --none)")
        cmd_next(a)
    elif a.cmd == "status":
        cmd_status(a)
    else:
        today = d(a.today) if a.today else dt.datetime.now(ET).date()
        build(a.root, today)


if __name__ == "__main__":
    main()
